22 Develop a menu-driven Employee Management System that manages employee records in 
Salesforce. 


| Field Label | Data Type | API Name       |
| ----------- | --------- | -------------- |
| Employee ID | Number    | Employee_ID__c |
| Department  | Text      | Department__c  |
| Salary      | Number    | Salary__c      |


apex class 

EmployeeManagementSystem 

public class EmployeeManagementSystem {

    // Object for form input
    public Employee__c employee {get; set;}

    // List to display employees
    public List<Employee__c> employeeList {get; set;}

    // Constructor
    public EmployeeManagementSystem() {

        employee = new Employee__c();

        loadEmployees();
    }

    // Load employee records
    public void loadEmployees() {

        employeeList = [
            SELECT Id,
                   Name,
                   Employee_ID__c,
                   Department__c,
                   Salary__c
            FROM Employee__c
        ];
    }

    // Save employee
    public PageReference saveEmployee() {

        insert employee;

        employee = new Employee__c();

        loadEmployees();

        return null;
    }

    // Delete employee
    public PageReference deleteEmployee() {

        String empId =
            ApexPages.currentPage()
                     .getParameters()
                     .get('id');

        Employee__c e = [
            SELECT Id
            FROM Employee__c
            WHERE Id = :empId
            LIMIT 1
        ];

        delete e;

        loadEmployees();

        return null;
    }
}

EmployeeManagementPage

<apex:page controller="EmployeeManagementSystem">

    <apex:form>
