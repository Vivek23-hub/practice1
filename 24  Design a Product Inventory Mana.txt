24  Design a Product Inventory Management System to maintain product records. 

| Field Label | Data Type | API Name      |
| ----------- | --------- | ------------- |
| Product ID  | Number    | Product_ID__c |
| Category    | Text      | Category__c   |
| Price       | Number    | Price__c      |
| Quantity    | Number    | Quantity__c   |


public class ProductInventoryManagementSystem {

    // Form object
    public Product_Inventory__c product {get; set;}

    // Product list
    public List<Product_Inventory__c> productList {get; set;}

    // Edit mode
    public Boolean isEdit {get; set;}

    // Constructor
    public ProductInventoryManagementSystem() {

        product = new Product_Inventory__c();

        isEdit = false;

        loadProducts();
    }

    // Load products
    public void loadProducts() {

        productList = [
            SELECT Id,
                   Name,
                   Product_ID__c,
                   Category__c,
                   Price__c,
                   Quantity__c
            FROM Product_Inventory__c
        ];
    }

    // Save product
    public PageReference saveProduct() {

        insert product;

        product = new Product_Inventory__c();

        loadProducts();

        return null;
    }

    // Edit product
    public PageReference editProduct() {

        String prodId =
            ApexPages.currentPage()
                     .getParameters()
                     .get('id');

        product = [
            SELECT Id,
                   Name,
                   Product_ID__c,
                   Category__c,
                   Price__c,
                   Quantity__c
            FROM Product_Inventory__c
            WHERE Id = :prodId
            LIMIT 1
        ];

        isEdit = true;

        return null;
    }

    // Update product
    public PageReference updateProduct() {

        update product;

        product = new Product_Inventory__c();

        isEdit = false;

        loadProducts();

        return null;
    }

    // Delete product
    public PageReference deleteProduct() {

        String prodId =
            ApexPages.currentPage()
                     .getParameters()
                     .get('id');

        Product_Inventory__c p = [
            SELECT Id
            FROM Product_Inventory__c
            WHERE Id = :prodId
            LIMIT 1
        ];

        delete p;

        loadProducts();

        return null;
    }
}
